```javascript
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';
import { test, fc } from '@fast-check/vitest';
import { StingerBar } from '../../src/stinger/StingerBar.js';
import { createMockCanvas, createMockContext } from '../fixtures/canvas-contexts.js';

describe('StingerBar phase transitions', () => {
  beforeEach(() => {
    vi.useFakeTimers({ toFake: ['requestAnimationFrame'] });
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  describe('lifecycle phases', () => {
    it('should progress through fadeIn -> visible -> fadeOut -> complete', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

      // Track phase progression
      const phases = [bar.phase];

      // Get durations from bar (converted from config)
      const fadeInDuration = bar.fadeInDuration;
      const visibleDuration = bar.visibleDuration;
      const fadeOutDuration = bar.fadeOutDuration;

      // Simulate through fadeIn phase
      let elapsed = 0;
      while (bar.phase === 'fadeIn' && elapsed < fadeInDuration + 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
        if (bar.phase !== phases[phases.length - 1]) {
          phases.push(bar.phase);
        }
      }

      // Should now be in visible phase
      expect(bar.phase).toBe('visible');

      // Simulate through visible phase
      elapsed = 0;
      while (bar.phase === 'visible' && elapsed < visibleDuration + 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
        if (bar.phase !== phases[phases.length - 1]) {
          phases.push(bar.phase);
        }
      }

      // Should now be in fadeOut phase
      expect(bar.phase).toBe('fadeOut');

      // Simulate through fadeOut phase
      elapsed = 0;
      while (bar.phase === 'fadeOut' && elapsed < fadeOutDuration + 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
        if (bar.phase !== phases[phases.length - 1]) {
          phases.push(bar.phase);
        }
      }

      // Should now be complete
      expect(bar.phase).toBe('complete');

      // Verify phase progression order
      expect(phases).toEqual(['fadeIn', 'visible', 'fadeOut', 'complete']);
    });

    it('should stop updating after reaching complete phase', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

      // Fast-forward to complete phase
      const totalDuration = bar.fadeInDuration + bar.visibleDuration + bar.fadeOutDuration;
      let elapsed = 0;
      while (elapsed < totalDuration + 1.0) {
        bar.update(0.016);
        elapsed += 0.016;
      }

      expect(bar.phase).toBe('complete');

      // Track position after complete
      const finalX = bar.x;

      // Continue updating - position should not change
      for (let i = 0; i < 60; i++) {
        bar.update(0.016);
      }

      expect(bar.x).toBeCloseTo(finalX, 2);
      expect(bar.phase).toBe('complete');
    });
  });

  describe('opacity curves', () => {
    it('should fade in from 0 to maxOpacity during fadeIn phase', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

      // Initial opacity should be 0
      expect(bar.currentOpacity).toBe(0);

      const maxOpacity = bar.maxOpacity;
      const fadeInDuration = bar.fadeInDuration;

      // Sample opacity throughout fadeIn
      const opacities = [];
      let elapsed = 0;
      while (bar.phase === 'fadeIn' && elapsed < fadeInDuration + 0.1) {
        opacities.push(bar.currentOpacity);
        bar.update(0.016);
        elapsed += 0.016;
      }

      // Verify opacity increased from 0 toward maxOpacity
      expect(opacities[0]).toBeCloseTo(0, 2);
      expect(opacities[opacities.length - 1]).toBeCloseTo(maxOpacity, 1);

      // Verify opacity is monotonically increasing
      for (let i = 1; i < opacities.length; i++) {
        expect(opacities[i]).toBeGreaterThanOrEqual(opacities[i - 1] - 0.01); // Allow tiny float variance
      }
    });

    it('should fade out from maxOpacity to 0 during fadeOut phase', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

      // Fast-forward to fadeOut phase
      const skipDuration = bar.fadeInDuration + bar.visibleDuration;
      let elapsed = 0;
      while (elapsed < skipDuration) {
        bar.update(0.016);
        elapsed += 0.016;
      }

      expect(bar.phase).toBe('fadeOut');

      const maxOpacity = bar.maxOpacity;

      // Sample opacity throughout fadeOut
      const opacities = [];
      elapsed = 0;
      while (bar.phase === 'fadeOut' && elapsed < bar.fadeOutDuration + 0.1) {
        opacities.push(bar.currentOpacity);
        bar.update(0.016);
        elapsed += 0.016;
      }

      // Verify opacity decreased from maxOpacity toward 0
      expect(opacities[0]).toBeCloseTo(maxOpacity, 1);
      expect(opacities[opacities.length - 1]).toBeCloseTo(0, 1);

      // Verify opacity is monotonically decreasing
      for (let i = 1; i < opacities.length; i++) {
        expect(opacities[i]).toBeLessThanOrEqual(opacities[i - 1] + 0.01); // Allow tiny float variance
      }
    });
  });

  describe('position bounds', () => {
    it('should maintain positions within reasonable canvas bounds throughout lifecycle', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

      // Coordinate system contract: Origin at top-left (0,0)
      // X increases rightward (0 to canvas.width)
      // Y increases downward (0 to canvas.height)
      // StingerBar uses inherited AnimatedBar positioning

      // Sample positions throughout full lifecycle
      const positions = [];
      const totalDuration = bar.fadeInDuration + bar.visibleDuration + bar.fadeOutDuration;
      let elapsed = 0;

      while (elapsed < totalDuration + 0.5 && bar.phase !== 'complete') {
        positions.push({ x: bar.x, y: bar.y, phase: bar.phase });
        bar.update(0.016);
        elapsed += 0.016;
      }

      // Verify all positions remain within reasonable bounds
      // StingerBar doesn't drift like BackgroundBar, so positions should stay near initial values
      const maxBound = Math.max(canvas.width, canvas.height) * 2; // Allow 2x canvas for off-screen rendering
      positions.forEach(pos => {
        expect(Math.abs(pos.x)).toBeLessThan(maxBound);
        expect(Math.abs(pos.y)).toBeLessThan(maxBound);
      });
    });
  });
});
```

```javascript
describe('StingerBar property-based tests', () => {
  test.prop([
    fc.float({ min: 0.5, max: 2.0 }) // speed multiplier
  ], { seed: 53, numRuns: 100 })(
    'phase transitions occur at expected times',
    (speed) => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, speed, false);

      const fadeInDuration = bar.fadeInDuration;
      const visibleDuration = bar.visibleDuration;
      const fadeOutDuration = bar.fadeOutDuration;

      // Simulate to just before fadeIn->visible transition
      let elapsed = 0;
      while (elapsed < fadeInDuration - 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
      }
      expect(bar.phase).toBe('fadeIn');

      // Advance past transition
      while (elapsed < fadeInDuration + 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
      }
      expect(bar.phase).toBe('visible');

      // Simulate to just before visible->fadeOut transition
      while (elapsed < fadeInDuration + visibleDuration - 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
      }
      expect(bar.phase).toBe('visible');

      // Advance past transition
      while (elapsed < fadeInDuration + visibleDuration + 0.1) {
        bar.update(0.016);
        elapsed += 0.016;
      }
      expect(bar.phase).toBe('fadeOut');
    }
  );

  test.prop([
    fc.float({ min: 0.5, max: 2.0 }),   // speed multiplier
    fc.float({ min: 1.0, max: 5.0 })    // elapsed time
  ], { seed: 54, numRuns: 100 })(
    'positions remain bounded during animation',
    (speed, elapsedTime) => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, speed, false);

      // Simulate animation
      const steps = Math.floor(elapsedTime / 0.016);
      for (let i = 0; i < steps; i++) {
        bar.update(0.016);
      }

      // Positions should remain within reasonable bounds
      const maxBound = Math.max(canvas.width, canvas.height) * 2;
      expect(Math.abs(bar.x)).toBeLessThan(maxBound);
      expect(Math.abs(bar.y)).toBeLessThan(maxBound);
    }
  );
});
```

**Note on position testing for StingerBar:**
StingerBar inherits AnimatedBar positioning and doesn't implement unique position logic like BackgroundBar's drift behavior. The position tests here validate that positions remain within reasonable bounds throughout the lifecycle, which is the appropriate contract for StingerBar. BackgroundBar has explicit position-locking tests because it has complex drift logic with direction changes - StingerBar does not.

Use toBeCloseTo(value, 2) for opacity comparisons (0.01 tolerance).

Use fixed seeds (53-54) for reproducibility and 100 runs per property test.
