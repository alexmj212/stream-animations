**Frame rate simulation approach:**
Mock DeltaTimer to return different delta values based on frame rate:
- 30fps: 0.033s per frame (1000ms / 30fps = 33.3ms)
- 60fps: 0.016s per frame (1000ms / 60fps = 16.6ms)
- 120fps: 0.008s per frame (1000ms / 120fps = 8.3ms)

**Frame rate consistency tests:**

```javascript
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';
import { test, fc } from '@fast-check/vitest';
import { BackgroundBar } from '../../src/background/BackgroundBar.js';
import { StingerBar } from '../../src/stinger/StingerBar.js';
import { SidebarBar } from '../../src/sidebar/SidebarBar.js';
import { createMockCanvas, createMockContext } from '../fixtures/canvas-contexts.js';

describe('Frame rate consistency', () => {
  beforeEach(() => {
    vi.useFakeTimers({ toFake: ['requestAnimationFrame'] });
  });

  afterEach(() => {
    vi.useRealTimers();
  });

  describe('BackgroundBar frame rate independence', () => {
    it('should complete visible phase in same time at 30fps, 60fps, and 120fps', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const frameRates = [
        { fps: 30, delta: 1 / 30 },   // 0.033s
        { fps: 60, delta: 1 / 60 },   // 0.016s
        { fps: 120, delta: 1 / 120 }  // 0.008s
      ];

      const results = frameRates.map(({ fps, delta }) => {
        const bar = new BackgroundBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

        // Fast-forward through fadeIn to visible phase
        let elapsed = 0;
        while (bar.phase !== 'visible' && elapsed < 10) {
          bar.update(delta);
          elapsed += delta;
        }

        const visibleStartTime = elapsed;

        // Simulate through visible phase
        while (bar.phase === 'visible' && elapsed < 30) {
          bar.update(delta);
          elapsed += delta;
        }

        const visibleEndTime = elapsed;
        const visibleDuration = visibleEndTime - visibleStartTime;

        return { fps, visibleDuration, finalPhase: bar.phase };
      });

      // All frame rates should complete visible phase in approximately same time
      const durations = results.map(r => r.visibleDuration);
      const avgDuration = durations.reduce((a, b) => a + b) / durations.length;

      // All durations should be within 5% of average (tolerance for frame timing)
      durations.forEach((duration, i) => {
        expect(duration).toBeCloseTo(avgDuration, 1); // 0.1s tolerance
        expect(results[i].finalPhase).toBe('fadeOut'); // Should all transition to fadeOut
      });
    });

    it('should reach same position at same elapsed time regardless of frame rate', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const targetTime = 2.0; // 2 seconds elapsed
      const frameRates = [
        { fps: 30, delta: 1 / 30 },
        { fps: 60, delta: 1 / 60 },
        { fps: 120, delta: 1 / 120 }
      ];

      const positions = frameRates.map(({ fps, delta }) => {
        const bar = new BackgroundBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

        // Simulate until target time
        let elapsed = 0;
        while (elapsed < targetTime) {
          bar.update(delta);
          elapsed += delta;
        }

        return { fps, x: bar.x, y: bar.y, phase: bar.phase };
      });

      // All frame rates should reach approximately same position
      const xPositions = positions.map(p => p.x);
      const yPositions = positions.map(p => p.y);

      // Compare positions - should be within tolerance
      expect(xPositions[0]).toBeCloseTo(xPositions[1], 1); // 30fps vs 60fps
      expect(xPositions[1]).toBeCloseTo(xPositions[2], 1); // 60fps vs 120fps
      expect(yPositions[0]).toBeCloseTo(yPositions[1], 1);
      expect(yPositions[1]).toBeCloseTo(yPositions[2], 1);

      // All should be in same phase
      const phases = positions.map(p => p.phase);
      expect(phases[0]).toBe(phases[1]);
      expect(phases[1]).toBe(phases[2]);
    });
  });

  describe('StingerBar frame rate independence', () => {
    it('should complete full lifecycle in same time at different frame rates', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const frameRates = [
        { fps: 30, delta: 1 / 30 },
        { fps: 60, delta: 1 / 60 },
        { fps: 120, delta: 1 / 120 }
      ];

      const results = frameRates.map(({ fps, delta }) => {
        const bar = new StingerBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

        // Simulate until complete phase
        let elapsed = 0;
        while (bar.phase !== 'complete' && elapsed < 30) {
          bar.update(delta);
          elapsed += delta;
        }

        return { fps, completionTime: elapsed, phase: bar.phase };
      });

      // All frame rates should complete in approximately same time
      const times = results.map(r => r.completionTime);
      const avgTime = times.reduce((a, b) => a + b) / times.length;

      times.forEach((time, i) => {
        expect(time).toBeCloseTo(avgTime, 1); // 0.1s tolerance
        expect(results[i].phase).toBe('complete');
      });
    });
  });

  describe('SidebarBar frame rate independence', () => {
    it('should hold for same duration at different frame rates', () => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const frameRates = [
        { fps: 30, delta: 1 / 30 },
        { fps: 60, delta: 1 / 60 },
        { fps: 120, delta: 1 / 120 }
      ];

      const results = frameRates.map(({ fps, delta }) => {
        const bar = new SidebarBar(ctx, canvas, 'horizontal', baseHSL, 1.0, false);

        // Fast-forward to hold phase
        let elapsed = 0;
        while (bar.phase !== 'hold' && elapsed < 10) {
          bar.update(delta);
          elapsed += delta;
        }

        const holdStartTime = elapsed;

        // Simulate through hold phase
        while (bar.phase === 'hold' && elapsed < 30) {
          bar.update(delta);
          elapsed += delta;
        }

        const holdEndTime = elapsed;
        const holdDuration = holdEndTime - holdStartTime;

        return { fps, holdDuration, phase: bar.phase };
      });

      // All frame rates should have same hold duration
      const durations = results.map(r => r.holdDuration);
      const avgDuration = durations.reduce((a, b) => a + b) / durations.length;

      durations.forEach((duration, i) => {
        expect(duration).toBeCloseTo(avgDuration, 1); // 0.1s tolerance
        expect(results[i].phase).toBe('slideOut'); // Should all transition to slideOut
      });
    });
  });
});
```

```javascript
describe('Frame rate consistency property tests', () => {
  test.prop([
    fc.integer({ min: 30, max: 120 }), // frame rate
    fc.float({ min: 1.0, max: 5.0 })   // target elapsed time
  ], { seed: 56, numRuns: 50 })( // Reduced runs due to integration test cost
    'BackgroundBar position at target time is frame-rate independent',
    (fps, targetTime) => {
      const canvas = createMockCanvas(1920, 1080);
      const ctx = createMockContext(canvas);
      const baseHSL = { h: 180, s: 50, l: 50 };

      const delta = 1 / fps;
      const bar = new BackgroundBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);

      // Simulate to target time
      let elapsed = 0;
      while (elapsed < targetTime) {
        bar.update(delta);
        elapsed += delta;
      }

      // Position should be deterministic based on elapsed time, not frame count
      // Re-run with double frame rate (half delta)
      const bar2 = new BackgroundBar(ctx, canvas, 'slanted', baseHSL, 1.0, false);
      const delta2 = 1 / (fps * 2);
      elapsed = 0;
      while (elapsed < targetTime) {
        bar2.update(delta2);
        elapsed += delta2;
      }

      // Positions should match within tolerance
      expect(bar.x).toBeCloseTo(bar2.x, 1);
      expect(bar.phase).toBe(bar2.phase);
    }
  );
});
```
