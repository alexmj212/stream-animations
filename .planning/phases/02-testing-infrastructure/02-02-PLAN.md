Create or extend tests/shared/timing.test.js with property-based tests for DeltaTimer invariants.

Critical invariants from Phase 1 decisions:
1. Delta values are always non-negative (no backward time)
2. Large deltas are capped at 0.1 seconds (100ms) to prevent tab-switch jumps
3. Null initialization handles timestamp=0 correctly
4. getDeltaTime() returns consistent values for same timestamp input

Property-based test structure:
```javascript
import { test, fc } from '@fast-check/vitest';
import { expect } from 'vitest';
import { DeltaTimer } from '../../src/shared/timing.js';

test.prop([
  fc.integer({ min: 0, max: 10000 }),  // timestamp1 (ms)
  fc.integer({ min: 0, max: 10000 })   // timestamp2 (ms)
], { seed: 42, numRuns: 100 })(
  'delta time is always non-negative',
  (t1, t2) => {
    const timer = new DeltaTimer();
    timer.getDeltaTime(t1);
    const delta = timer.getDeltaTime(t2);
    expect(delta).toBeGreaterThanOrEqual(0);
  }
);

test.prop([
  fc.integer({ min: 0, max: 1000 }),   // initial timestamp
  fc.integer({ min: 200, max: 10000 }) // large delta (>100ms)
], { seed: 43, numRuns: 100 })(
  'large deltas are capped at 0.1 seconds',
  (t1, t2) => {
    const timer = new DeltaTimer();
    timer.getDeltaTime(t1);
    const delta = timer.getDeltaTime(t1 + t2);
    expect(delta).toBeLessThanOrEqual(0.1);
  }
);
```

Also test null initialization handling (timestamp=0) and consecutive calls returning consistent deltas.

Use fixed seed (42, 43, etc.) for reproducibility and 100 runs per property for thorough coverage.

Create or extend tests/shared/cache.test.js with property-based tests for LRUCache eviction behavior.

Critical invariants from Phase 1:
1. Cache size never exceeds maxSize (LRU eviction works)
2. Stale entries are evicted based on maxAgeMs
3. get() updates access time (LRU tracking)
4. Cache operations are deterministic for same input sequence

Property-based test structure:
```javascript
import { test, fc } from '@fast-check/vitest';
import { expect } from 'vitest';
import { LRUCache } from '../../src/shared/cache.js';

test.prop([
  fc.integer({ min: 1, max: 10 }),      // maxSize
  fc.array(fc.string(), { minLength: 1, maxLength: 50 }) // keys to insert
], { seed: 44, numRuns: 100 })(
  'cache size never exceeds maxSize',
  (maxSize, keys) => {
    const cache = new LRUCache(maxSize);
    keys.forEach((key, i) => cache.set(key, i));
    expect(cache.size).toBeLessThanOrEqual(maxSize);
  }
);

test.prop([
  fc.integer({ min: 100, max: 1000 }),  // maxAgeMs
  fc.integer({ min: 0, max: 2000 }),    // elapsed time
  fc.string()                            // cache key
], { seed: 45, numRuns: 100 })(
  'stale entries are evicted by evictStale',
  (maxAgeMs, elapsedMs, key) => {
    const cache = new LRUCache(10);
    const now = Date.now();

    // Mock Date.now for predictable time
    const originalNow = Date.now;
    Date.now = () => now;

    cache.set(key, 'value');

    // Advance time
    Date.now = () => now + elapsedMs;
    cache.evictStale(maxAgeMs);

    // Restore Date.now
    Date.now = originalNow;

    if (elapsedMs > maxAgeMs) {
      expect(cache.has(key)).toBe(false);
    } else {
      expect(cache.has(key)).toBe(true);
    }
  }
);
```

Test both size-based eviction (maxSize) and time-based eviction (maxAgeMs). Verify LRU order is maintained (least recently used evicted first).

Use fixed seeds for reproducibility and 100 runs per property.

Create or extend tests/shared/lifecycle.test.js with property-based tests for Lifecycle cleanup behavior.

Critical invariants from Phase 1:
1. All event listeners added via addListener() are removed on cleanup()
2. AbortController is aborted exactly once per cleanup
3. Multiple cleanup() calls are idempotent (no errors)
4. Cleanup works regardless of listener count (0 to many)

Property-based test structure:
```javascript
import { test, fc } from '@fast-check/vitest';
import { expect, vi } from 'vitest';
import { Lifecycle } from '../../src/shared/lifecycle.js';

test.prop([
  fc.integer({ min: 0, max: 20 }) // number of listeners
], { seed: 46, numRuns: 100 })(
  'all event listeners are removed on cleanup',
  (listenerCount) => {
    const lifecycle = new Lifecycle();
    const mockElement = {
      addEventListener: vi.fn(),
      removeEventListener: vi.fn()
    };

    // Add multiple listeners
    for (let i = 0; i < listenerCount; i++) {
      const handler = vi.fn();
      lifecycle.addListener(mockElement, 'event' + i, handler);
    }

    // Cleanup should remove all
    lifecycle.cleanup();

    // Verify AbortController was aborted (listeners cleaned up)
    expect(lifecycle.aborted).toBe(true);
  }
);

test.prop([
  fc.integer({ min: 1, max: 5 }) // number of cleanup calls
], { seed: 47, numRuns: 100 })(
  'multiple cleanup calls are idempotent',
  (cleanupCount) => {
    const lifecycle = new Lifecycle();
    const mockElement = { addEventListener: vi.fn() };
    lifecycle.addListener(mockElement, 'test', () => {});

    // Call cleanup multiple times
    for (let i = 0; i < cleanupCount; i++) {
      expect(() => lifecycle.cleanup()).not.toThrow();
    }

    expect(lifecycle.aborted).toBe(true);
  }
);
```

Test that cleanup is reliable regardless of listener count, event types, or number of cleanup invocations.

Use fixed seeds for reproducibility and 100 runs per property.
