```javascript
/**
 * Shared test fixtures for canvas-based tests
 */

/**
 * Create mock canvas with specified dimensions
 * @param {number} width - Canvas width (default: 1920)
 * @param {number} height - Canvas height (default: 1080)
 * @returns {HTMLCanvasElement} Canvas element
 */
export function createMockCanvas(width = 1920, height = 1080) {
  const canvas = document.createElement('canvas');
  canvas.width = width;
  canvas.height = height;
  return canvas;
}

/**
 * Create mock 2D rendering context
 * @param {HTMLCanvasElement} canvas - Canvas element
 * @returns {CanvasRenderingContext2D} 2D context
 */
export function createMockContext(canvas) {
  return canvas.getContext('2d');
}

/**
 * Create canvas, context, and bar for testing
 * @param {string} type - Bar type ('slanted' or 'horizontal')
 * @param {Object} options - Optional configuration
 * @returns {Object} { canvas, ctx, bar }
 */
export function createMockBar(BarClass, type = 'slanted', options = {}) {
  const canvas = createMockCanvas(options.width, options.height);
  const ctx = createMockContext(canvas);
  const bar = new BarClass(ctx, canvas, type, options);
  return { canvas, ctx, bar };
}
```

This fixture file will be reused across all bar tests (AnimatedBar, BackgroundBar, StingerBar, SidebarBar).

Following CONTEXT.md decision: "Create helpers only if patterns repeat 3+ times across tests" - canvas creation is used in every bar test file, so this is justified.

**Property-based tests:**
```javascript
import { test, fc } from '@fast-check/vitest';
import { describe, it, expect } from 'vitest';
import { AnimatedBar } from '../../src/shared/AnimatedBar.js';
import { createMockCanvas, createMockContext } from '../fixtures/canvas-contexts.js';

describe('AnimatedBar', () => {
  describe('easing function', () => {
    test.prop([
      fc.float({ min: 0, max: 1 }) // t value
    ], { seed: 48, numRuns: 100 })(
      'easeInOutCubic returns value in 0-1 range for input 0-1',
      (t) => {
        const canvas = createMockCanvas();
        const ctx = createMockContext(canvas);
        const bar = new AnimatedBar(ctx, canvas, 'slanted');

        const result = bar.easeInOutCubic(t);

        // Invariant: output must be in 0-1 range for valid input
        expect(result).toBeGreaterThanOrEqual(0);
        expect(result).toBeLessThanOrEqual(1);
      }
    );

    test.prop([
      fc.float({ min: 0, max: 1 }) // t value
    ], { seed: 49, numRuns: 100 })(
      'easeInOutCubic is monotonic (increasing)',
      (t1) => {
        const canvas = createMockCanvas();
        const ctx = createMockContext(canvas);
        const bar = new AnimatedBar(ctx, canvas, 'slanted');

        const t2 = Math.min(1, t1 + 0.1); // t2 > t1
        const result1 = bar.easeInOutCubic(t1);
        const result2 = bar.easeInOutCubic(t2);

        // Invariant: easing function should be increasing
        expect(result2).toBeGreaterThanOrEqual(result1);
      }
    );
  });

  describe('offscreen cache', () => {
    test.prop([
      fc.integer({ min: 10, max: 100 }), // width
      fc.integer({ min: 10, max: 100 }), // height
      fc.integer({ min: 0, max: 360 }),  // hue
      fc.integer({ min: 0, max: 100 }),  // saturation
      fc.integer({ min: 0, max: 100 })   // lightness
    ], { seed: 50, numRuns: 100 })(
      'cache key is deterministic for same properties',
      (width, height, hue, sat, light) => {
        const canvas = createMockCanvas();
        const ctx = createMockContext(canvas);
        const bar = new AnimatedBar(ctx, canvas, 'slanted');

        const canvas1 = bar.getOrCreateOffscreenCanvas(width, height, hue, sat, light, 15, false);
        const canvas2 = bar.getOrCreateOffscreenCanvas(width, height, hue, sat, light, 15, false);

        // Invariant: same inputs return same cached canvas
        expect(canvas1).toBe(canvas2);
      }
    );
  });
});
```
