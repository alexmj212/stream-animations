```javascript
/**
 * Validate duration values (milliseconds)
 * @param {object} durations - Duration config object
 * @param {object} defaults - Default values
 * @returns {object} Validated durations
 */
export function validateDurations(durations, defaults) {
  if (!durations || typeof durations !== 'object') {
    console.warn('Config validation: durations must be an object. Using defaults.');
    return defaults;
  }

  return {
    defaultMs: validateNumber(durations.defaultMs, defaults.defaultMs, 'durations.defaultMs', 100, 10000),
    entryMs: validateNumber(durations.entryMs, defaults.entryMs, 'durations.entryMs', 100, 10000),
    exitMs: validateNumber(durations.exitMs, defaults.exitMs, 'durations.exitMs', 100, 10000),
    holdMs: validateNumber(durations.holdMs, defaults.holdMs, 'durations.holdMs', 0, 30000)
  };
}

/**
 * Validate brightness multiplier (0-1 range)
 * @param {number} brightness - Brightness multiplier
 * @param {number} defaultBrightness - Default value
 * @param {string} label - Config path for logging
 * @returns {number} Validated brightness (0-1)
 */
export function validateBrightness(brightness, defaultBrightness, label) {
  return validateNumber(brightness, defaultBrightness, label, 0, 1);
}

/**
 * Validate easing function name exists
 * @param {string} easingFunction - Function name
 * @param {string} defaultEasing - Default value
 * @returns {string} Validated easing function name
 */
export function validateEasingFunction(easingFunction, defaultEasing = 'easeInOutCubic') {
  if (typeof easingFunction !== 'string' || easingFunction.trim() === '') {
    console.warn(`Config validation: easingFunction must be a non-empty string. Using default: ${defaultEasing}`);
    return defaultEasing;
  }
  // Note: Currently only easeInOutCubic implemented in AnimatedBar
  // This validates format only, not that method exists
  return easingFunction;
}

/**
 * Validate sidebar positioning constraints
 * @param {object} position - Position config
 * @param {object} defaults - Default values
 * @returns {object} Validated position
 */
export function validateSidebarPositioning(position, defaults) {
  if (!position || typeof position !== 'object') {
    console.warn('Config validation: sidebar.position must be an object. Using defaults.');
    return defaults;
  }

  const sidebarStart = validateNumber(position.sidebarStart, defaults.sidebarStart, 'sidebar.position.sidebarStart', 0, 1);
  const sidebarWidth = validateNumber(position.sidebarWidth, defaults.sidebarWidth, 'sidebar.position.sidebarWidth', 0, 1);

  // Logical constraint: sidebar must fit within viewport
  if (sidebarStart + sidebarWidth > 1.0) {
    console.warn(
      `Config validation: sidebar.position.sidebarStart (${sidebarStart}) + sidebarWidth (${sidebarWidth}) exceeds viewport (1.0). Sidebar may extend off-screen.`
    );
  }

  return {
    sidebarStart,
    sidebarWidth,
    horizontalTargetOffset: validateNumber(
      position.horizontalTargetOffset,
      defaults.horizontalTargetOffset,
      'sidebar.position.horizontalTargetOffset',
      -1000,
      1000
    )
  };
}
```

describe('duration validation', () => {
  test('validateDurations accepts valid millisecond values', () => {
    const durations = { defaultMs: 1000, entryMs: 1500, exitMs: 800, holdMs: 2000 };
    const result = validateDurations(durations, { defaultMs: 1000, entryMs: 1000, exitMs: 1000, holdMs: 2000 });
    expect(result).toEqual(durations);
  });

  test('validateDurations clamps values outside range', () => {
    const durations = { defaultMs: 50, entryMs: 20000, exitMs: 1000, holdMs: 50000 };
    const defaults = { defaultMs: 1000, entryMs: 1000, exitMs: 1000, holdMs: 2000 };
    const result = validateDurations(durations, defaults);

    expect(result.defaultMs).toBe(1000); // 50 < min(100)
    expect(result.entryMs).toBe(1000);   // 20000 > max(10000)
    expect(result.holdMs).toBe(2000);    // 50000 > max(30000)
  });

  test('validateDurations returns defaults for invalid input', () => {
    const defaults = { defaultMs: 1000, entryMs: 1000, exitMs: 1000, holdMs: 2000 };
    expect(validateDurations(null, defaults)).toEqual(defaults);
    expect(validateDurations('invalid', defaults)).toEqual(defaults);
  });
});

describe('brightness validation', () => {
  test('validateBrightness accepts values in 0-1 range', () => {
    expect(validateBrightness(0.8, 0.8, 'test')).toBe(0.8);
    expect(validateBrightness(0, 0.8, 'test')).toBe(0);
    expect(validateBrightness(1, 0.8, 'test')).toBe(1);
  });

  test('validateBrightness clamps values outside 0-1', () => {
    expect(validateBrightness(-0.5, 0.8, 'test')).toBe(0.8); // < 0
    expect(validateBrightness(1.5, 0.8, 'test')).toBe(0.8);  // > 1
  });

  test('validateBrightness returns default for non-numeric input', () => {
    expect(validateBrightness('0.8', 0.8, 'test')).toBe(0.8);
    expect(validateBrightness(null, 0.8, 'test')).toBe(0.8);
  });
});

describe('easing function validation', () => {
  test('validateEasingFunction accepts valid string', () => {
    expect(validateEasingFunction('easeInOutCubic')).toBe('easeInOutCubic');

  });

  test('validateEasingFunction returns default for invalid input', () => {
    expect(validateEasingFunction('')).toBe('easeInOutCubic');
    expect(validateEasingFunction(null)).toBe('easeInOutCubic');
