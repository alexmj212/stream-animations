Create ShadeBar class extending AnimatedBar following established pattern from StingerBar/SidebarBar.

**Class structure:**
```javascript
export class ShadeBar extends AnimatedBar {
  constructor(ctx, canvas, text, color, holdDurationMs) {
    super(ctx, canvas, 'text', {});
    this.text = text;
    this.color = color; // Explicit color for OBS compatibility (TEXT-01)
    this.holdDurationMs = holdDurationMs;

    // Text positioning
    this.x = canvas.width; // Start offscreen right (from CONTEXT.md)
    this.y = canvas.height / 2; // Vertical center

    // Text metrics (will be measured after font loads)
    this.textWidth = 0;

    // Phase tracking (for future Phase 06 integration)
    this.phase = 'enter'; // 'enter' | 'hold' | 'exit'
    this.elapsedTime = 0;
  }

  init() {
    // Measure text width (call this AFTER font loading in index.html)
    this.ctx.font = config.shade.text.font;
    const metrics = this.ctx.measureText(this.text);
    this.textWidth = metrics.width;
  }

  update(deltaTime) {
    // Stub for Phase 06 - will implement scrolling animation in Plan 02
    this.elapsedTime += deltaTime;
  }

  draw() {
    this.ctx.save();

    // Set text properties
    this.ctx.font = config.shade.text.font;
    this.ctx.fillStyle = this.color; // Explicit fillStyle for OBS (RESEARCH.md Pitfall 2)
    this.ctx.textBaseline = 'middle';

    // Draw shadow/glow (from CONTEXT.md and RESEARCH.md)
    this.ctx.shadowColor = config.shade.text.shadow.color;
    this.ctx.shadowBlur = config.shade.text.shadow.blur;
    this.ctx.shadowOffsetX = config.shade.text.shadow.offsetX;
    this.ctx.shadowOffsetY = config.shade.text.shadow.offsetY;

    // Draw text
    this.ctx.fillText(this.text, this.x, this.y);

    this.ctx.restore();
  }
}
```

**Important:**
- Use explicit fillStyle (not relying on defaults) - OBS browser source requirement (RESEARCH.md Pitfall 2)
- Measure text AFTER font loading (not in constructor) - prevents race condition (RESEARCH.md Pitfall 1)
- Position text offscreen right initially (x = canvas.width) - ready for enter animation in Plan 02
- Keep draw() simple for now - no scrolling logic yet (that's Plan 02)

Follow established conventions: PascalCase filename (ShadeBar.js), named export, JSDoc comments.

```javascript
// Import dependencies
import { ShadeBar } from './ShadeBar.js';
import { getColorFromURL } from '../shared/colors.js';
import { config } from '../config/config.js';
import { DeltaTimer } from '../shared/timing.js';

// Parse URL parameters (CONFIG-01)
const urlParams = new URLSearchParams(window.location.search);
const text = urlParams.get('text') || 'Default Text';
const color = getColorFromURL(); // Handles ?color or generates random
const durationParam = urlParams.get('duration');
const holdDurationMs = durationParam ? parseInt(durationParam) : config.shade.timing.minHoldMs;

// Setup canvas
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d', { alpha: config.performance.alpha.background });
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

// Wait for fonts before initialization (RESEARCH.md Pitfall 1)
document.fonts.ready.then(() => {
  const shadeBar = new ShadeBar(ctx, canvas, text, color, holdDurationMs);
  shadeBar.init(); // Measure text after fonts loaded

  const deltaTimer = new DeltaTimer();

  function animate(timestamp) {
    const deltaTime = deltaTimer.update(timestamp);

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    shadeBar.update(deltaTime);
    shadeBar.draw();

    requestAnimationFrame(animate);
  }

  requestAnimationFrame(animate);
});
```
