update(deltaTime) {

  const phaseTime = this.elapsedTime - this.phaseStartTime;

  // Update position based on phase
  if (this.phase === 'in') {
    const progress = Math.min(phaseTime / this.inDuration, 1);
    const easedProgress = this.easeInOutCubic(progress);
    this.x = this.startX + (this.targetX - this.startX) * easedProgress;

Update update(deltaTime) method:

update(deltaTime) {
  if (!this.active) return;

  this.elapsedTime += deltaTime;

  // Handle delay before animation starts
  if (this.elapsedTime < this.delayTime) {
    this.x = this.startX; // Stay offscreen
    return;
  }

  const animTime = this.elapsedTime - this.delayTime;
  const phaseTime = animTime - this.phaseStartTime;

  if (this.phase === 'in') {
    const progress = Math.min(phaseTime / this.slideInDuration, 1);
    const easedProgress = this.easeInOutCubic(progress);
    this.x = this.startX + (this.holdX - this.startX) * easedProgress;

    if (progress >= 1) {
      this.phase = 'hold';
      this.phaseStartTime = animTime;
    }
  } else if (this.phase === 'hold') {
    this.x = this.holdX;

    if (phaseTime >= this.holdDuration) {
      this.phase = 'out';
      this.phaseStartTime = animTime;
    }
  } else if (this.phase === 'out') {
    const progress = Math.min(phaseTime / this.slideOutDuration, 1);
    const easedProgress = this.easeInOutCubic(progress);
    this.x = this.holdX + (this.endX - this.holdX) * easedProgress;

    if (progress >= 1) {
      this.active = false;
    }
  }
}
