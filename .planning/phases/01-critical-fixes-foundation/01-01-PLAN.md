Create `src/shared/timing.js` implementing DeltaTimer class for frame-rate independent animations.

**Key implementation points from RESEARCH.md:**
- Use requestAnimationFrame timestamp parameter (DOMHighResTimeStamp in milliseconds)
- Calculate delta in seconds for smoother values: `(timestamp - lastTimestamp) / 1000`
- Cap delta-time to 0.1 seconds (100ms) to prevent tab-switch jumps
- Store lastTimestamp to track between frames
- Provide update(timestamp) method that returns capped deltaTime

**Implementation:**
```javascript
export class DeltaTimer {
  constructor() {
    this.lastTimestamp = 0;
    this.maxDeltaTime = 0.1; // 100ms cap prevents tab-switch spikes
  }

  /**
   * Update timer with new timestamp from requestAnimationFrame
   * @param {DOMHighResTimeStamp} timestamp - Timestamp from rAF
   * @returns {number} Delta time in seconds (capped)
   */
  update(timestamp) {
    if (this.lastTimestamp === 0) {
      this.lastTimestamp = timestamp;
      return 0; // First frame has no delta
    }

    const rawDelta = (timestamp - lastTimestamp) / 1000;
    const deltaTime = Math.min(rawDelta, this.maxDeltaTime);
    this.lastTimestamp = timestamp;

    return deltaTime;
  }

  reset() {
    this.lastTimestamp = 0;
  }
}
```

**What to avoid:**
- Do NOT use Date.now() - use only timestamp parameter from requestAnimationFrame
- Do NOT use setInterval/setTimeout for animation timing
- Do NOT use frame counting (frameCount++) for time-based logic

  <action>
Create `src/shared/cache.js` implementing LRUCache class for managing AnimatedBar offscreen canvas cache.

**Key implementation points from RESEARCH.md:**
- Use native Map (maintains insertion order, O(1) operations)
- Track access times with separate Map for time-based eviction
- Implement both size-based (LRU) and time-based (stale) eviction
- Move accessed items to end (most recently used) by delete + re-set
- Conservative defaults: maxSize=50, maxAgeMs=60000 (60 seconds)

**Implementation pattern from RESEARCH.md lines 136-196:**
```javascript
export class LRUCache {
  constructor(maxSize = 50) {
    this.cache = new Map();
    this.accessTimes = new Map();
    this.maxSize = maxSize;
  }

  get(key) {
    if (!this.cache.has(key)) return undefined;

    // Update access time
    this.accessTimes.set(key, Date.now());

    // Move to end (most recently used)
    const value = this.cache.get(key);
    this.cache.delete(key);
    this.cache.set(key, value);

    return value;
  }

  set(key, value) {
    // Delete if exists (to update position)
    if (this.cache.has(key)) {
      this.cache.delete(key);
    }

    // Add to end (most recently used)
    this.cache.set(key, value);
    this.accessTimes.set(key, Date.now());

    // Evict oldest if over size
    if (this.cache.size > this.maxSize) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
      this.accessTimes.delete(firstKey);
    }
  }

  // Time-based cleanup: remove entries unused for X milliseconds
  evictStale(maxAgeMs = 60000) {
    const now = Date.now();
    for (const [key, accessTime] of this.accessTimes) {
      if (now - accessTime > maxAgeMs) {
        this.cache.delete(key);
        this.accessTimes.delete(key);
      }
    }
  }

  clear() {
    this.cache.clear();
    this.accessTimes.clear();
  }

  get size() {
    return this.cache.size;
  }
}
```

**What to avoid:**
- Do NOT use external dependencies (lru-cache package) - use native Map
- Do NOT track memory usage directly (too much overhead) - only track cache.size
- Do NOT use WeakMap (entries never garbage collected while references exist)

```javascript
export class AnimationLifecycle {
  constructor() {
    this.controller = new AbortController();
    this.signal = this.controller.signal;
    this.cleanupCallbacks = [];
    this.setupBeforeUnload();
  }

  /**
   * Add event listener that will be auto-cleaned on abort
   * @param {EventTarget} target - Element to attach listener to
   * @param {string} event - Event name
   * @param {Function} handler - Event handler
   * @param {Object} options - Additional options (merged with signal)
   */
  addEventListener(target, event, handler, options = {}) {
    target.addEventListener(event, handler, {
      ...options,
      signal: this.signal
    });
  }

  /**
   * Register callback to run on cleanup
   * @param {Function} callback - Cleanup function
   */
  onCleanup(callback) {
    this.cleanupCallbacks.push(callback);
  }

  /**
   * Clean up all listeners and run cleanup callbacks
   */
  cleanup() {
    // Run custom cleanup callbacks
    for (const callback of this.cleanupCallbacks) {
      try {
        callback();
      } catch (error) {
        console.error('Cleanup callback error:', error);
      }
    }

    // Abort all event listeners
    this.controller.abort();
    console.log('[Lifecycle] All event listeners cleaned up');
  }

  setupBeforeUnload() {
    // Use once:true so this listener removes itself
    window.addEventListener('beforeunload', () => {
      this.cleanup();
    }, { once: true });
  }
}
```

**What to avoid:**
- Do NOT store function references for removeEventListener - use AbortController
- Do NOT create multiple AbortControllers per listener - use single shared instance
- Do NOT ignore cleanup callback errors - catch and log them
