import { config } from '../config/config.js';

// At class level (line 6)

static cacheCleanupInterval = null;

// Add static initialization method
static initializeCache() {
  // Setup periodic cache cleanup if not already running
  if (!AnimatedBar.cacheCleanupInterval) {
    AnimatedBar.cacheCleanupInterval = setInterval(() => {
      AnimatedBar.offscreenCache.evictStale(config.background.cache.maxAgeMs);

      // Log cache stats in dev mode only

      if (urlParams.get('perf') === 'true') {
        console.log(`[Cache] Size: ${AnimatedBar.offscreenCache.size}/${config.background.cache.maxSize}`);
      }
    }, config.background.cache.evictionIntervalMs);
  }
}
